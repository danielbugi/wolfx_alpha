# Database (sync + async pools — see shared/database.py)

asyncpg==0.29.0

# Data fetching
# NOTE: yfinance is unpinned deliberately. Yahoo Finance changes its
# internal (unofficial) API frequently, and yfinance ships fixes as patch/
# minor releases. Pinning to 0.2.40 (the original version here) is exactly
# what caused the pipeline to go dark for ~10 months (2025-11 to 2026-09) —
# it returned "possibly delisted" for every symbol until upgraded to 1.7.0.
# Run `pip install --upgrade yfinance` if data fetching starts failing again.
yfinance>=1.7.0
requests==2.31.0
beautifulsoup4==4.12.3

# Interactive Telegram bot (mechanism/alerts/run_bot.py, added 2026-09-21). The daily broadcast
# (send_daily_digest.py) only needs `requests`; aiogram is required just for the always-on bot process.
aiogram>=3.13,<4

# Market-performance card image (mechanism/alerts/market_card.py, added 2026-09-21). Pillow only; the DejaVu font is bundled
# under mechanism/alerts/assets/fonts/ so the card renders identically on any machine.
pillow>=10.1

# Per-stock candlestick chart of the private assistant (mechanism/alerts/chart.py, added 2026-09-21), rendered locally from stock_prices.
# Verified with matplotlib 3.8.3 + mplfinance 0.12.10b0 (mplfinance is a beta line; keep the pin loose but >= 0.12.10).
matplotlib>=3.8
mplfinance>=0.12.10b0

# Optional alternate price provider (shared/alpaca_client.py). Inert unless
